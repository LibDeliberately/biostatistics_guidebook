Summarizing Averages

# summary measurements like mean will return NA when NA values are present in a column.
# We can fix this by omitting the NA variables for now
clean_penguins = na.omit(penguins)

clean_penguins %>%
  summarize(
    mean = mean(bill_length_mm),
    sd = sd(bill_length_mm),
    median = median(bill_length_mm),
    min = min(bill_length_mm),
    max = max(bill_length_mm)
  )
# A tibble: 1 × 5
   mean    sd median   min   max
  <dbl> <dbl>  <dbl> <dbl> <dbl>
1  44.0  5.47   44.5  32.1  59.6
# We can also group this by a categorical variable

clean_penguins %>%
  group_by(sex) %>%
  summarize(
    mean = mean(bill_length_mm),
    sd = sd(bill_length_mm),
    median = median(bill_length_mm),
    min = min(bill_length_mm),
    max = max(bill_length_mm)
  )
